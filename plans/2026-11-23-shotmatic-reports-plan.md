**Tech Stack:** Python 3 (venv `.venv`), Pillow ≥ 10.1, PyYAML, pytest, MkDocs 1.6 + mkdocs-material 9, GitHub Actions Pages.

```python
#!/usr/bin/env python3
"""Stamp the ShotMatic badge on a screenshot and save it under its report name."""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont, UnidentifiedImageError

ROOT = Path(__file__).resolve().parent.parent
LOGO = ROOT / "docs" / "images" / "shotmatic-wordmark-light.png"
MAX_WIDTH = 2400
BADGE_TEXT = "shotmatic.app"
BADGE_BG = (31, 42, 68, 224)  # navy #1F2A44 at ~88%
FONT_CANDIDATES = [
    "/System/Library/Fonts/Helvetica.ttc",
    "/System/Library/Fonts/SFNS.ttf",
    "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
]


class WatermarkError(Exception):
    pass


def image_name(slug: str, date: str) -> str:
    return f"shotmatic.app-{slug}-{date}.png"


def badge_height(img_h: int) -> int:
    return max(28, min(64, round(img_h * 0.04)))


def _font(size: int) -> ImageFont.FreeTypeFont:
    for p in FONT_CANDIDATES:
        if Path(p).exists():
            try:
                return ImageFont.truetype(p, size)
            except OSError:
                continue
    return ImageFont.load_default(size=size)


def make_badge(height: int, logo_path: Path = LOGO) -> Image.Image:
    logo = Image.open(logo_path).convert("RGBA")
    lh = round(height * 0.62)
    lw = round(logo.width * lh / logo.height)
    logo = logo.resize((lw, lh), Image.LANCZOS)

    font = _font(round(height * 0.42))
    left, top, right, bottom = font.getbbox(BADGE_TEXT)
    tw, th = right - left, bottom - top
    pad, gap = round(height * 0.3), round(height * 0.28)
    width = pad + lw + gap + tw + pad

    badge = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    draw = ImageDraw.Draw(badge)
    draw.rounded_rectangle((0, 0, width - 1, height - 1), radius=height // 3, fill=BADGE_BG)
    badge.alpha_composite(logo, (pad, (height - lh) // 2))
    draw.text((pad + lw + gap - left, (height - th) // 2 - top), BADGE_TEXT,
              font=font, fill=(255, 255, 255, 255))
    return badge


def watermark(src: Path, out_dir: Path, slug: str, date: str, logo_path: Path = LOGO) -> Path:
    try:
        img = Image.open(src)
        img.load()
    except (UnidentifiedImageError, OSError) as e:
        raise WatermarkError(f"cannot open image {src}: {e}") from e

    img = img.convert("RGBA")
    if img.width > MAX_WIDTH:
        img = img.resize((MAX_WIDTH, round(img.height * MAX_WIDTH / img.width)), Image.LANCZOS)

    badge = make_badge(badge_height(img.height), logo_path)
    margin = max(8, round(min(img.size) * 0.02))
    x, y = img.width - badge.width - margin, img.height - badge.height - margin
    if x < margin or y < margin:
        raise WatermarkError(f"image too small for the badge: {img.width}x{img.height}")
    img.alpha_composite(badge, (x, y))

    out_dir.mkdir(parents=True, exist_ok=True)
    out = out_dir / image_name(slug, date)
    img.convert("RGB").save(out, "PNG", optimize=True)
    return out


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("src", type=Path)
    ap.add_argument("--out-dir", type=Path, required=True)
    ap.add_argument("--slug", required=True)
    ap.add_argument("--date", required=True)
    a = ap.parse_args(argv)
    try:
        print(watermark(a.src, a.out_dir, a.slug, a.date))
    except WatermarkError as e:
        print(f"error: {e}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

#!/usr/bin/env python3
"""Create a report post (stamped image + skeleton) and regenerate nav / index pages."""
from __future__ import annotations

import argparse
import datetime as dt
import json
import re
import sys
import unicodedata
from pathlib import Path

import yaml

from watermark import WatermarkError, watermark

ROOT = Path(__file__).resolve().parent.parent
FM_KEYS = ["title", "date", "type", "keyword", "short", "image", "source", "x_caption"]
NAV_BEGIN = "# BEGIN NAV (generated by scripts/new_post.py — do not edit by hand)"
NAV_END = "# END NAV"
POSTS_BEGIN = "<!-- BEGIN POSTS -->"
POSTS_END = "<!-- END POSTS -->"

BODY = """# {title}

## The short answer

WRITE: 40–60 words that answer the question a searcher has about "{keyword}", with 1–2 numbers from the screenshot.

![{keyword} — ShotMatic report]({image})

## What the data shows

- WRITE: 3–5 bullets, each with a number and where it comes from.

## What this means if you build in this niche

WRITE: 2–3 sentences of advice for an indie app maker.

## How this was measured

{source}.

[Try ShotMatic free](https://shotmatic.app){{ .md-button .md-button--primary }}
"""


def load_types(root: Path = ROOT) -> dict[str, str]:
    return json.loads((root / "scripts" / "types.json").read_text())


def slugify(text: str, max_len: int = 60) -> str:
    ascii_text = unicodedata.normalize("NFKD", text).encode("ascii", "ignore").decode()
    slug = re.sub(r"[^a-z0-9]+", "-", ascii_text.lower()).strip("-")
    if len(slug) > max_len:
        cut = slug[:max_len]
        slug = cut.rsplit("-", 1)[0] if "-" in cut and slug[max_len] != "-" else cut
    return slug.strip("-")


def read_post(path: Path) -> tuple[dict, str]:
    text = path.read_text()
    if not text.startswith("---\n"):
        raise ValueError(f"{path}: missing frontmatter")
    _, fm_text, body = text.split("---\n", 2)
    fm = yaml.safe_load(fm_text) or {}
    if "date" in fm:
        fm["date"] = str(fm["date"])
    return fm, body


def _write_post(path: Path, fm: dict, body: str) -> None:
    fm_text = yaml.safe_dump(fm, sort_keys=False, allow_unicode=True, width=1000)
    path.write_text(f"---\n{fm_text}---\n\n{body}")


def _unique_slug(root: Path, type_: str, date: str, slug: str) -> str:
    folder = root / "docs" / type_
    candidate, n = slug, 2
    while (folder / f"{date}-{candidate}.md").exists():
        candidate, n = f"{slug}-{n}", n + 1
    return candidate


def create_post(root: Path, type_: str, date: str, slug: str, title: str,
                keyword: str, source: str, image_src: Path) -> dict:
    if type_ not in load_types(root):
        raise ValueError(f"unknown type {type_!r}; add it to scripts/types.json")
    try:
        dt.date.fromisoformat(date)
    except ValueError as e:
        raise ValueError(f"date must be YYYY-MM-DD, got {date!r}") from e
    slug = _unique_slug(root, type_, date, slugify(slug))
    folder = root / "docs" / type_
    image = watermark(Path(image_src), folder / "img", slug, date,
                      root / "docs" / "images" / "shotmatic-wordmark-light.png")
    rel_image = f"img/{image.name}"
    fm = {
        "title": title,
        "date": date,
        "type": type_,
        "keyword": keyword,
        "short": slug.replace("-", " ").capitalize(),
        "image": rel_image,
        "source": source,
        "x_caption": "",
    }
    post = folder / f"{date}-{slug}.md"
    _write_post(post, fm, BODY.format(title=title, keyword=keyword, image=rel_image, source=source.rstrip(".")))
    rebuild_site(root)
    return {"slug": slug, "post": str(post), "image": str(image)}


def _posts_by_type(root: Path) -> dict[str, list[tuple[dict, Path]]]:
    out: dict[str, list[tuple[dict, Path]]] = {}
    for type_ in load_types(root):
        items = []
        for p in sorted((root / "docs" / type_).glob("*.md")):
            if p.name == "index.md":
                continue
            fm, _ = read_post(p)
            items.append((fm, p))
        items.sort(key=lambda it: (it[0]["date"], it[1].name), reverse=True)
        if items:
            out[type_] = items
    return out


def _short_date(d: str) -> str:
    x = dt.date.fromisoformat(d)
    return f"{x:%b} {x.day}"


def _long_date(d: str) -> str:
    x = dt.date.fromisoformat(d)
    return f"{x:%b} {x.day}, {x.year}"


def _replace_block(text: str, begin: str, end: str, inner: str) -> str:
    head, rest = text.split(begin, 1)
    _, tail = rest.split(end, 1)
    return f"{head}{begin}\n{inner}{end}{tail}"


def rebuild_site(root: Path = ROOT) -> None:
    types = load_types(root)
    by_type = _posts_by_type(root)

    nav = ["nav:", "  - Home: index.md"]
    for type_, items in by_type.items():
        nav.append(f"  - {types[type_]}:")
        nav.append(f"      - {type_}/index.md")
        for fm, p in items:
            label = json.dumps(f"{_short_date(fm['date'])} · {fm['short']}", ensure_ascii=False)
            nav.append(f"      - {label}: {type_}/{p.name}")
    yml_path = root / "mkdocs.yml"
    yml_path.write_text(_replace_block(yml_path.read_text(), NAV_BEGIN, NAV_END, "\n".join(nav) + "\n"))

    all_posts = sorted(
        ((fm, p, type_) for type_, items in by_type.items() for fm, p in items),
        key=lambda it: (it[0]["date"], it[1].name), reverse=True,
    )
    lines, current = [], None
    for fm, p, type_ in all_posts:
        if fm["date"] != current:
            current = fm["date"]
            lines += ["", f"### {_long_date(current)}", ""]
        lines.append(f"- [{fm['title']}]({type_}/{p.name}) · {types[type_]}")
    inner = ("\n".join(lines).lstrip("\n") if lines else "_No reports yet._") + "\n"
    home = root / "docs" / "index.md"
    home.write_text(_replace_block(home.read_text(), POSTS_BEGIN, POSTS_END, inner))

    for type_, items in by_type.items():
        body = [f"# {types[type_]} reports", ""]
        body += [f"- {_long_date(fm['date'])} · [{fm['title']}]({p.name})" for fm, p in items]
        (root / "docs" / type_ / "index.md").write_text("\n".join(body) + "\n")


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    sub = ap.add_subparsers(dest="cmd", required=True)
    c = sub.add_parser("create")
    c.add_argument("--type", required=True)
    c.add_argument("--slug", required=True)
    c.add_argument("--title", required=True)
    c.add_argument("--keyword", required=True)
    c.add_argument("--source", required=True)
    c.add_argument("--image", type=Path, required=True)
    c.add_argument("--date", default=dt.date.today().isoformat())
    sub.add_parser("nav")
    a = ap.parse_args(argv)
    try:
        if a.cmd == "create":
            r = create_post(ROOT, a.type, a.date, a.slug, a.title, a.keyword, a.source, a.image)
            print(json.dumps(r, indent=2))
        else:
            rebuild_site(ROOT)
    except (ValueError, WatermarkError) as e:
        print(f"error: {e}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

```python
#!/usr/bin/env python3
"""Lint report posts before they are pushed."""
from __future__ import annotations

import re
import sys
from pathlib import Path

from new_post import FM_KEYS, load_types, read_post

ROOT = Path(__file__).resolve().parent.parent
URL_RE = re.compile(r"https?://\S+")
POST_NAME_RE = re.compile(r"^(\d{4}-\d{2}-\d{2})-[a-z0-9-]+\.md$")
IMAGE_RE = re.compile(r"^img/shotmatic\.app-[a-z0-9-]+-\d{4}-\d{2}-\d{2}\.png$")
SHORT_RE = re.compile(r"^## The short answer\s*\n(.*?)(?=^## |\Z)", re.M | re.S)


def x_length(caption: str) -> int:
    return len(URL_RE.sub("x" * 23, caption))


def short_answer(body: str) -> str | None:
    m = SHORT_RE.search(body)
    return m.group(1) if m else None


def check_post(path: Path, root: Path = ROOT) -> list[str]:
    try:
        fm, body = read_post(path)
    except Exception as e:  # malformed YAML or no frontmatter
        return [f"cannot read frontmatter: {e}"]

    errors = [f"missing or empty frontmatter: {k}" for k in FM_KEYS if not str(fm.get(k, "")).strip()]

    type_ = fm.get("type")
    if type_ not in load_types(root):
        errors.append(f"type {type_!r} is not in scripts/types.json")
    elif path.parent.resolve() != (root / "docs" / type_).resolve():
        errors.append(f"post must live in folder docs/{type_}/")

    m = POST_NAME_RE.match(path.name)
    if not m:
        errors.append("file name must be YYYY-MM-DD-<slug>.md")
    elif m.group(1) != fm.get("date"):
        errors.append(f"file name date {m.group(1)} != frontmatter date {fm.get('date')}")

    image = str(fm.get("image", ""))
    if image:
        if not IMAGE_RE.match(image):
            errors.append("image name must be img/shotmatic.app-<slug>-<YYYY-MM-DD>.png")
        if not (path.parent / image).exists():
            errors.append(f"image not found: {image}")

    if "WRITE:" in body:
        errors.append("unfilled WRITE: placeholder left in the post")

    sa = short_answer(body)
    if sa is None:
        errors.append("missing '## The short answer' section")
    else:
        n = len(sa.split())
        if not 40 <= n <= 80:
            errors.append(f"short answer is {n} words; keep it 40–80 (target 40–60)")

    if "## How this was measured" not in body:
        errors.append("missing '## How this was measured' section")

    caption = str(fm.get("x_caption", ""))
    if caption and x_length(caption) > 280:
        errors.append(f"x_caption is {x_length(caption)} chars; X allows 280")
    return errors


def all_posts(root: Path = ROOT) -> list[Path]:
    posts = []
    for type_ in load_types(root):
        posts += [p for p in sorted((root / "docs" / type_).glob("*.md")) if p.name != "index.md"]
    return posts


def main(argv: list[str] | None = None) -> int:
    args = sys.argv[1:] if argv is None else argv
    posts = [Path(a) for a in args] or all_posts()
    if not posts:
        print("no posts")
        return 0
    failed = False
    for p in posts:
        errors = check_post(p)
        if errors:
            failed = True
            print(f"FAIL {p}")
            for e in errors:
                print(f"  - {e}")
        else:
            print(f"OK {p}")
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main())
```
